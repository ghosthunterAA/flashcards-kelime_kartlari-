        attackFlags[myTeam] = List.filled(5, true);
        skipTeamNextTurn[myTeam] = false;
      }

      if (pile[currentSeat].isEmpty && hand[currentSeat].isEmpty) {
        isGameOver = true;
        winnerTeam = oppTeam;
        matchLog = "🃏 ${seatLabel(currentSeat)} çekecek kartı kalmadı! ${teamLabel(oppTeam)} kazandı!";
        return;
      }
      _drawOne(currentSeat);
      final String attackNote = turnNumber <= 1 ? " (bu tur saldırı yok)" : "";
      matchLog = "🔄 Sıra ${seatLabel(currentSeat)}'de.$attackNote (${hand[currentSeat].length} kart elinde)";
    });

    if (isGameOver || !mounted) return;
    _startSeatTurn();
  }

  void _startSeatTurn() {
    if (currentIsBot) {
      _scheduleBotTurn();
    } else {
      _showHandover(currentSeat, first: turnNumber <= 1 && currentSeat == 0);
    }
  }

  Future<void> _enforceHandLimit(int seat) async {
    final h = hand[seat];
    while (h.length > tdHandLimit) {
      dynamic chosen;
      if (widget.seats[seat].isBot) {
        final monsters = h.whereType<CharacterCardModel>().toList();
        if (monsters.isNotEmpty) {
          monsters.sort((a, b) => _tdHandValue(a).compareTo(_tdHandValue(b)));
          chosen = monsters.first;
        } else {
          chosen = h.first;
        }
      } else {
        chosen = await _askDiscardChoice(seat, h);
        if (!mounted) return;
      }
      setState(() {
        h.remove(chosen);
        if (chosen is CharacterCardModel) {
          grave[seat < 2 ? 0 : 1].add(chosen);
        }
      });
    }
  }

  // ---------------- BOT ----------------
  void _scheduleBotTurn() {
    setState(() => _botBusy = true);
    Future.delayed(const Duration(milliseconds: 700), () async {
      if (!mounted || isGameOver) return;
      _botTrapStep();
      await Future.delayed(const Duration(milliseconds: 400));
      if (!mounted || isGameOver) return;
      _botSummonStep();
      await Future.delayed(const Duration(milliseconds: 500));
      if (!mounted || isGameOver) return;
      _botFlipStep();
      await Future.delayed(const Duration(milliseconds: 400));
      if (!mounted || isGameOver) return;
      await _botStoneStep();
      await Future.delayed(const Duration(milliseconds: 400));
      if (!mounted || isGameOver) return;
      await _botAttackStep();
      if (!mounted) return;
      setState(() => _botBusy = false);
      await Future.delayed(const Duration(milliseconds: 500));
      if (!mounted) return;
      _endTurn();
    });
  }

  void _botTrapStep() {
    final double chance = currentBotLevel == BotLevel.easy ? 0.25 : (currentBotLevel == BotLevel.normal ? 0.55 : 0.8);
    for (int n = 0; n < 5; n++) {
      final traps = hand[currentSeat].whereType<SpellTrapCardModel>().where((c) => c.kind == STKind.trap || c.effect == STEffect.negateTrap).toList();
      if (traps.isEmpty) return;
      final int slot = _mySt.indexOf(null);
      if (slot == -1) return;
      if (Random().nextDouble() > chance) return;
      setState(() {
        final c = traps.first;
        _mySt[slot] = c;
        hand[currentSeat].remove(c);
        matchLog = "🃏 ${seatLabel(currentSeat)} kapalı bir kart kurdu.";
      });
    }
  }

  void _botSummonStep() {
    final level = currentBotLevel;
    final List<CharacterCardModel> monsters = hand[currentSeat].whereType<CharacterCardModel>().toList();
    final candidates = monsters.where((c) => _canSummon(_myField, c.stars)).toList();
    if (candidates.isEmpty) return;

    bool decideDefense(CharacterCardModel c) {
      if (level == BotLevel.easy) return Random().nextDouble() < 0.3;
      final bool lowLife = hp[myTeam] < hp[oppTeam] * 0.6 || hp[myTeam] < 12000;
      final int oppBest = [for (int i = 0; i < 5; i++) if (_oppField[i] != null) _effAtk(_oppField[i]!)].fold(0, (a, b) => a > b ? a : b);
      final bool weakerThanOpp = oppBest > 0 && c.atk < oppBest;
      final bool naturallyDefensive = c.def > c.atk * 1.3;
      if (level == BotLevel.hard) return lowLife || weakerThanOpp || naturallyDefensive;
      return (lowLife || naturallyDefensive) && Random().nextDouble() < 0.6;
    }

    if (level == BotLevel.easy) {
      if (Random().nextDouble() < 0.6) {
        final c = candidates[Random().nextInt(candidates.length)];
        final need = _tributesNeeded(c.stars);
        final tribs = [for (int i = 0; i < 5; i++) if (_myField[i] != null) i]..shuffle();
        _placeMonsterDirect(myTeam, currentSeat, c, tribs.take(need).toList(), decideDefense(c));
      }
      return;
    }

    final double lam = level == BotLevel.hard ? 0.6 : 1.0;
    double bestGain = 0;
    CharacterCardModel? bestCard;
    List<int> bestTribs = [];
    for (final c in candidates) {
      final need = _tributesNeeded(c.stars);
      final ownSlots = [for (int i = 0; i < 5; i++) if (_myField[i] != null) i]
        ..sort((a, b) => _tdHandValue(_myField[a]!.card).compareTo(_tdHandValue(_myField[b]!.card)));
      final tribs = ownSlots.take(need).toList();
      final double lost = tribs.fold(0.0, (s, i) => s + _tdHandValue(_myField[i]!.card));
      final double gain = _tdHandValue(c) - lost * lam;
      if (gain > bestGain) {
        bestGain = gain;
        bestCard = c;
        bestTribs = tribs;
      }
    }
    if (bestCard != null) _placeMonsterDirect(myTeam, currentSeat, bestCard, bestTribs, decideDefense(bestCard));
  }

  void _botFlipStep() {
    if (flipUsedThisTurn || currentBotLevel == BotLevel.easy) return;
    final oppSlots = [for (int i = 0; i < 5; i++) if (_oppField[i] != null) i];

    for (int i = 0; i < 5; i++) {
      final m = _myField[i];
      if (m == null || !m.isDefense || _myAttackFlags[i]) continue;
      final int atkVal = _effAtk(m);
      bool worth = oppSlots.isEmpty;
      for (final j in oppSlots) {
        final DuelMonster d = _oppField[j]!;
        final int dv = d.isDefense ? _effDef(d) : _effAtk(d);
        if (atkVal > dv) worth = true;
      }
      if (worth) {
        setState(() {
          m.isDefense = false;
          m.faceDown = false;
          flipUsedThisTurn = true;
          matchLog = "🔓 ${seatLabel(currentSeat)} bir canavarını saldırı pozisyonuna çevirdi!";
        });
        return;
      }
    }

    final bool lowLife = hp[myTeam] < hp[oppTeam] * 0.6 || hp[myTeam] < 12000;
    if (lowLife) {
      final int oppBest = oppSlots.map((j) => _oppField[j]!.isDefense ? 0 : _effAtk(_oppField[j]!)).fold(0, (a, b) => a > b ? a : b);
      if (oppBest > 0) {
        for (int i = 0; i < 5; i++) {
          final m = _myField[i];
          if (m == null || m.isDefense || _myAttackFlags[i]) continue;
          if (_effAtk(m) < oppBest && _effDef(m) > _effAtk(m)) {
            setState(() {
              m.isDefense = true;
              flipUsedThisTurn = true;
              matchLog = "🛡️ ${seatLabel(currentSeat)} geri çekildi, bir canavarını savunmaya aldı!";
            });
            return;
          }
        }
      }
    }
  }

  Future<void> _botStoneStep() async {
    if (!_mySt.contains(null)) return;
    final owned = [for (int i = 0; i < 5; i++) if (_myField[i] != null && !_myField[i]!.faceDown) i];
    if (owned.isEmpty) return;
    final stones = hand[currentSeat].whereType<StoneCardModel>().toList();
    if (stones.isEmpty) return;
    final stone = stones.first;
    int best = owned.first;
    int bestAtk = -1;
    for (final i in owned) {
      final v = _effAtk(_myField[i]!);
      if (v > bestAtk) {
        bestAtk = v;
        best = i;
      }
    }
    await _attachStone(stone, best);
  }

  Future<void> _botAttackStep() async {
    if (turnNumber <= 1) return;
    final level = currentBotLevel;

    for (int atk = 0; atk < 5; atk++) {
      final DuelMonster? m = _myField[atk];
      if (m == null || m.isDefense || _myAttackFlags[atk]) continue;

      final oppSlots = [for (int i = 0; i < 5; i++) if (_oppField[i] != null) i];

      if (oppSlots.isEmpty) {
        await _resolveAttack(atk, null);
        continue;
      }
      if (level == BotLevel.easy && Random().nextDouble() > 0.5) continue;

      int? bestDef;
      double bestValue = level == BotLevel.hard ? 0 : -1e9;
      final int atkVal = _effAtk(m);
      for (final def in oppSlots) {
        if (_oppField[def] == null) continue;
        final DuelMonster d = _oppField[def]!;
        double v;
        if (d.faceDown) {
          final int guessDef = 900 + Random().nextInt(1300);
          v = atkVal > guessDef ? _tdHandValue(d.card) * 0.4 : -0.4 * (guessDef - atkVal);
        } else if (d.isDefense) {
          final int dv = _effDef(d);
          v = atkVal > dv ? _tdHandValue(d.card) : -0.3 * (dv - atkVal);
        } else {
          final int dv = _effAtk(d);
          if (atkVal > dv) {
            v = _tdHandValue(d.card) + (atkVal - dv) * 0.2;
          } else if (atkVal < dv) {
            v = -_tdHandValue(m.card) - (dv - atkVal) * 0.3;
          } else {
            v = _tdHandValue(d.card) - _tdHandValue(m.card);
          }
        }
        if (v > bestValue) {
          bestValue = v;
          bestDef = def;
        }
      }
      if (bestDef != null || level != BotLevel.hard) {
        await _resolveAttack(atk, bestDef);
      }
    }
  }

    final zone = st[team];

  }

        title: Text("${card.name} (${card.kind == STKind.spell ? 'Büyü' : 'Tuzak'})", textAlign: TextAlign.center, style: const TextStyle(color: Color(0xFFc5a059), fontSize: 15)),
        children: [
          SimpleDialogOption(onPressed: () { Navigator.pop(ctx); _showBigPreview(card); }, child: const Text("🔍 İncele", style: TextStyle(color: Colors.white))),
          if (card.kind == STKind.spell && card.effect != STEffect.negateTrap && !card.isEquip && _isMyInteractiveTurn)
            SimpleDialogOption(onPressed: () { Navigator.pop(ctx); _activateSetSpell(index); }, child: const Text("⚡ Aktifleştir", style: TextStyle(color: Colors.greenAccent))),
          SimpleDialogOption(onPressed: () => Navigator.pop(ctx), child: const Text("Kapat", style: TextStyle(color: Colors.grey))),
        ],
      ),
    );
  }

  void _onTapMyMonster(int index) {
    if (!_isMyInteractiveTurn || isGameOver) return;
    final m = _myField[index];
    if (m == null) return;
    final bool canAttack = !m.isDefense && !_myAttackFlags[index] && turnNumber > 1;
    final bool canFlip = !flipUsedThisTurn && !_myAttackFlags[index];
    showDialog<void>(
      context: context,
      builder: (ctx) => SimpleDialog(
        backgroundColor: const Color(0xFF18181f),
        title: Text(m.faceDown ? "❓ Kapalı Canavar" : "${m.card.name}\nATK ${_effAtk(m)} / DEF ${_effDef(m)}", textAlign: TextAlign.center, style: const TextStyle(color: Color(0xFFc5a059), fontSize: 15)),
        children: [
          if (m.faceDown) SimpleDialogOption(onPressed: () { Navigator.pop(ctx); _showBigPreview(m.card); }, child: const Text("🔍 İncele", style: TextStyle(color: Colors.white))),
          if (canAttack) SimpleDialogOption(onPressed: () { Navigator.pop(ctx); setState(() => selectedAttackerIndex = index); }, child: const Text("⚔️ Saldırgan Olarak Seç", style: TextStyle(color: Colors.white))),
          if (canFlip) SimpleDialogOption(onPressed: () { Navigator.pop(ctx); _tryFlip(index); }, child: const Text("🔄 Pozisyon Değiştir", style: TextStyle(color: Colors.white))),
          SimpleDialogOption(onPressed: () => Navigator.pop(ctx), child: const Text("İptal", style: TextStyle(color: Colors.grey))),
        ],
      ),
    );
  }

  Widget _infoButton(dynamic item) {
    return Positioned(
      top: 4, right: 10,
      child: GestureDetector(
        onTap: () => _showBigPreview(item),
        child: Container(padding: const EdgeInsets.all(4), decoration: const BoxDecoration(color: Colors.black87, shape: BoxShape.circle), child: const Icon(Icons.remove_red_eye, color: Colors.amberAccent, size: 16)),
      ),
    );
  }

  Widget _handCard(dynamic item) {
    if (item is SpellTrapCardModel) {
      return Padding(padding: const EdgeInsets.symmetric(horizontal: 6), child: Stack(children: [GestureDetector(onTap: () => _tryPlaySpellOrTrap(item), child: SpellTrapCardWidget(card: item, size: handW * 0.9)), _infoButton(item)]));
    }
    if (item is StoneCardModel) {
      return Padding(padding: const EdgeInsets.symmetric(horizontal: 6), child: Stack(children: [GestureDetector(onTap: () => _tryPlayStone(item), child: StoneCardWidget(stone: item, size: handW * 0.9)), _infoButton(item)]));
    }
    final CharacterCardModel c = item as CharacterCardModel;
    final bool ok = _canSummon(_myField, c.stars) && !summonedThisTurn;
    return Padding(
      padding: const EdgeInsets.symmetric(horizontal: 6),
      child: Stack(children: [
        GestureDetector(
          onTap: () => _trySummon(c),
          child: Container(
            width: handW, height: handH,
            decoration: BoxDecoration(color: const Color(0xFF1c1c24), border: Border.all(color: ok ? const Color(0xFFd4af37) : Colors.white24, width: 2.5), borderRadius: BorderRadius.circular(12)),
            child: Column(mainAxisSize: MainAxisSize.min, children: [
              SizedBox(
                height: 110, width: double.infinity,
                child: Stack(children: [
                  Positioned.fill(child: c.imagePath != null && File(c.imagePath!).existsSync() ? ClipRRect(borderRadius: const BorderRadius.vertical(top: Radius.circular(10)), child: Image.file(File(c.imagePath!), fit: BoxFit.cover)) : Container(color: const Color(0xFF101014), child: const Icon(Icons.image_not_supported, color: Colors.white24, size: 36))),
                  Positioned(
                    top: 3, left: 3,
                    child: Container(
                      padding: const EdgeInsets.all(3),
                      decoration: const BoxDecoration(color: Colors.black87, shape: BoxShape.circle),
                      child: Row(mainAxisSize: MainAxisSize.min, children: [Text(c.gender, style: TextStyle(fontSize: 11, fontWeight: FontWeight.bold, color: _genderColor(c.gender))), const SizedBox(width: 2), alignmentBadge(c.alignment, size: 11)]),
                    ),
                  ),
                ]),
              ),
              const SizedBox(height: 4),
              Text(c.name, maxLines: 1, overflow: TextOverflow.ellipsis, style: const TextStyle(fontSize: 13, color: Colors.white, fontWeight: FontWeight.bold)),
              Text("⭐${c.stars} · ${_tributesNeeded(c.stars)} feda", style: TextStyle(fontSize: 10, color: ok ? Colors.greenAccent : Colors.redAccent)),
              const SizedBox(height: 3),
              Text("ATK ${c.atk}  /  DEF ${c.def}", style: const TextStyle(fontSize: 14, color: Color(0xFFc5a059), fontWeight: FontWeight.bold)),
            ]),
          ),
        ),
        _infoButton(c),
      ]),
    );
  }
