import 'dart:convert';
import 'dart:typed_data';
import 'package:flutter/material.dart';
import 'package:image_picker/image_picker.dart';
import 'package:shared_preferences/shared_preferences.dart';

void main() {
  runApp(const MaterialApp(
    debugShowCheckedModeBanner: false,
    home: KelimeKartlariAnaSayfa(),
  ));
}

class KelimeKartlariAnaSayfa extends StatefulWidget {
  const KelimeKartlariAnaSayfa({super.key});

  @override
  State<KelimeKartlariAnaSayfa> createState() => _KelimeKartlariAnaSayfaState();
}

class _KelimeKartlariAnaSayfaState extends State<KelimeKartlariAnaSayfa> {
  List<Map<String, String>> _kelimeler = [];
  int _guncelEndeks = 0;
  bool _kartDondu = false;
  
  String? _secilenGorselVerisi; 
  bool _urlModuMu = false;

  final TextEditingController _ingilizceController = TextEditingController();
  final TextEditingController _urlController = TextEditingController();
  final ImagePicker _picker = ImagePicker();

  @override
  void initState() {
    super.initState();
    _verileriYukle();
  }

  Future<void> _verileriYukle() async {
    final prefs = await SharedPreferences.getInstance();
    final String? kelimelerString = prefs.getString('kalici_kelime_listesi');
    
    if (kelimelerString != null) {
      final List<dynamic> decodedList = jsonDecode(kelimelerString);
      setState(() {
        _kelimeler = decodedList.map((item) => Map<String, String>.from(item)).toList();
      });
    }
  }

  Future<void> _verileriKaydet() async {
    final prefs = await SharedPreferences.getInstance();
    final String encodedList = jsonEncode(_kelimeler);
    await prefs.setString('kalici_kelime_listesi', encodedList);
  }

  Future<void> _galeridenResimSec(StateSetter dialogState) async {
    final XFile? image = await _picker.pickImage(source: ImageSource.gallery);
    if (image != null) {
      final Uint8List imageBytes = await image.readAsBytes();
      dialogState(() {
        _secilenGorselVerisi = base64Encode(imageBytes);
      });
    }
  }

  Future<void> _kartSil() async {
    if (_kelimeler.isEmpty) return;

    setState(() {
      _kelimeler.removeAt(_guncelEndeks);
      _kartDondu = false;
      if (_guncelEndeks >= _kelimeler.length && _kelimeler.isNotEmpty) {
        _guncelEndeks = _kelimeler.length - 1;
      } else if (_kelimeler.isEmpty) {
        _guncelEndeks = 0;
      }
    });

    await _verileriKaydet();
  }

  void _yeniKartEkleDialog() {
    _secilenGorselVerisi = null;
    _urlModuMu = false;
    _ingilizceController.clear();
    _urlController.clear();

    showDialog(
      context: context,
      builder: (context) {
        return StatefulBuilder(
          builder: (context, dialogState) {
            return AlertDialog(
              title: const Text('Yeni Kart Ekle 🗺️'),
              content: SingleChildScrollView(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Row(
                      children: [
                        Expanded(
                          child: ElevatedButton(
                            style: ElevatedButton.styleFrom(
                              backgroundColor: !_urlModuMu ? Colors.blue : Colors.grey[800],
                            ),
                            onPressed: () {
                              dialogState(() {
                                _urlModuMu = false;
                                _secilenGorselVerisi = null;
                              });
                            },
                            child: const Text('Yerel Dosya'),
                          ),
                        ),
                        const SizedBox(width: 8),
                        Expanded(
                          child: ElevatedButton(
                            style: ElevatedButton.styleFrom(
                              backgroundColor: _urlModuMu ? Colors.blue : Colors.grey[800],
                            ),
                            onPressed: () {
                              dialogState(() {
                                _urlModuMu = true;
                                _secilenGorselVerisi = null;
                              });
                            },
                            child: const Text('URL Linki'),
                          ),
                        ),
                      ],
                    ),
                    const SizedBox(height: 20),

                    if (!_urlModuMu) ...[
                      ElevatedButton.icon(
                        onPressed: () => _galeridenResimSec(dialogState),
                        icon: const Icon(Icons.photo_library),
                        label: const Text('Galeriden Resim Seç'),
                      ),
                      const SizedBox(height: 10),
                      if (_secilenGorselVerisi != null)
                        const Text('🟢 Resim Yüklendi', style: TextStyle(color: Colors.green))
                      else
                        const Text('🔴 Resim Seçilmedi', style: TextStyle(color: Colors.red)),
                    ] else ...[
                      TextField(
                        controller: _urlController,
                        decoration: const InputDecoration(
                          labelText: 'Resim URL Adresi',
                          hintText: 'https://...',
                        ),
                      ),
                    ],

                    const SizedBox(height: 20),
                    TextField(
                      controller: _ingilizceController,
                      decoration: const InputDecoration(labelText: 'İngilizce Kelime'),
                    ),
                  ],
                ),
              ),
              actions: [
                TextButton(
                  onPressed: () => Navigator.pop(context),
                  child: const Text('İptal'),
                ),
                TextButton(
                  onPressed: () async {
                    String gorselDegeri = _urlModuMu ? _urlController.text : (_secilenGorselVerisi ?? '');

                    if (_ingilizceController.text.isNotEmpty && gorselDegeri.isNotEmpty) {
                      setState(() {
                        _kelimeler.add({
                          'ingilizce': _ingilizceController.text,
                          'gorsel': gorselDegeri,
                          'tip': _urlModuMu ? 'url' : 'base64',
                        });
                      });
                      await _verileriKaydet();
                      _ingilizceController.clear();
                      _urlController.clear();
                      
                      if (mounted) {
                        Navigator.pop(context);
                      }
                    }
                  },
                  child: const Text('Ekle'),
                ),
              ],
            );
          },
        );
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    final guncelKart = _kelimeler.isNotEmpty ? _kelimeler[_guncelEndeks] : null;

    Widget gorselWidgetGoster() {
      if (guncelKart == null) return const SizedBox();

      String gorselVerisi = guncelKart['gorsel']!;
      String tip = guncelKart['tip'] ?? 'base64';

      if (tip == 'url') {
        return Image.network(
          gorselVerisi,
          fit: BoxFit.cover,
          width: double.infinity,
          height: double.infinity,
          errorBuilder: (c, e, s) => const Text('URL Resim yüklenemedi.'),
        );
      } else {
        return Image.memory(
          base64Decode(gorselVerisi),
          fit: BoxFit.cover,
          width: double.infinity,
          height: double.infinity,
          errorBuilder: (c, e, s) => const Text('Yerel Resim yüklenemedi.'),
        );
      }
    }

    return Scaffold(
      appBar: AppBar(
        title: const Text('Kelime Kartlarım (Fiziksel Kalıcı Sürüm) 🚀'),
        actions: [
          if (_kelimeler.isNotEmpty)
            IconButton(
              icon: const Icon(Icons.delete, color: Colors.redAccent),
              onPressed: _kartSil,
              tooltip: 'Bu Kartı Sil',
            ),
          IconButton(
            icon: const Icon(Icons.add),
            onPressed: _yeniKartEkleDialog,
            tooltip: 'Yeni Kart Ekle',
          ),
        ],
      ),
      body: Center(
        child: guncelKart == null
            ? const Text('Henüz kart eklemediniz. Sağ üstten ekleyin.')
            : Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  GestureDetector(
                    onTap: () {
                      setState(() {
                        _kartDondu = !_kartDondu;
                      });
                    },
                    child: Container(
                      width: 300,
                      height: 300,
                      alignment: Alignment.center,
                      decoration: BoxDecoration(
                        border: Border.all(color: Colors.blue),
                        borderRadius: BorderRadius.circular(10),
                      ),
                      child: _kartDondu
                          ? Text(
                              guncelKart['ingilizce']!,
                              style: const TextStyle(fontSize: 32),
                            )
                          : gorselWidgetGoster(),
                    ),
                  ),
                  const SizedBox(height: 20),
                  ElevatedButton(
                    onPressed: () {
                      setState(() {
                        _guncelEndeks = (_guncelEndeks + 1) % _kelimeler.length;
                        _kartDondu = false;
                      });
                    },
                    child: const Text('Sonraki Kelime'),
                  ),
                ],
              ),
      ),
    );
  }
}
